  public class KadaneAlgorithm {
    public static void main(String[] args) {
        int[] arr = {-2, 1, -3, 4, -1, 2, 1, -5, 4};

        int maxSum = findMaxSubarraySum(arr);
        System.out.println("The maximum contiguous subarray sum is: " + maxSum);
    }

    public static int findMaxSubarraySum(int[] arr) {
        if (arr == null || arr.length == 0) {
            return 0;
        }

        int currentSum = arr[0];
        int maxSum = arr[0];

        // Iterate through the array starting from the second element
        for (int i = 1; i < arr.length; i++) {
            // Decide whether to add the current element to the existing subarray
            // or start a fresh subarray at the current element
            currentSum = Math.max(arr[i], currentSum + arr[i]);

            // Update the global maximum sum if the current subarray sum is greater
            maxSum = Math.max(maxSum, currentSum);
        }

        return maxSum;
    }
}
