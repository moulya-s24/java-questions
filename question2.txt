import java.util.Arrays;

public class DuplicateFinderWithoutMap {
    public static void main(String[] args) {
        int[] arr = {4, 3, 2, 7, 8, 2, 3, 1};

        System.out.print("Elements that appear more than once: ");
        boolean foundDuplicate = false;

        // Outer loop to pick each element
        for (int i = 0; i < arr.length; i++) {
            boolean alreadyChecked = false;
            
            // Check if this element was already processed as a duplicate earlier
            for (int k = 0; k < i; k++) {
                if (arr[k] == arr[i]) {
                    alreadyChecked = true;
                    break;
                }
            }
            
            if (alreadyChecked) {
                continue; // Skip if we already printed or checked this number
            }

            // Count occurrences of arr[i] in the array
            int count = 0;
            for (int j = 0; j < arr.length; j++) {
                if (arr[j] == arr[i]) {
                    count++;
                }
            }

            // If it appears more than once, print it
            if (count > 1) {
                System.out.print(arr[i] + " ");
                foundDuplicate = true;
            }
        }

        if (!foundDuplicate) {
            System.out.print("None");
        }
        System.out.println();
    }
}
