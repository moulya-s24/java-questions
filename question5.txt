public class MissingNumberFinder {
    public static void main(String[] args) {
        // Example array containing numbers from 1 to 5, with 3 missing
        int[] arr = {1, 2, 4, 5};
        int N = 5; // The array should contain integers from 1 to N

        int missingNumber = findMissingNumber(arr, N);
        System.out.println("The missing number is: " + missingNumber);
    }

    public static int findMissingNumber(int[] arr, int N) {
        // Step 1: Calculate the expected sum of numbers from 1 to N
        int expectedSum = N * (N + 1) / 2;

        // Step 2: Calculate the actual sum of elements in the array
        int actualSum = 0;
        for (int i = 0; i < arr.length; i++) {
            actualSum += arr[i];
        }

        // Step 3: The difference is the missing number
        return expectedSum - actualSum;
    }
}
